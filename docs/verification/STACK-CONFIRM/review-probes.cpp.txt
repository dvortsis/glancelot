    void probeSlideshowOverUndecodablePhoto() {
        Media media(3, QSize(64, 48));
        { QFile bad(media.path(1)); QVERIFY(bad.open(QIODevice::WriteOnly | QIODevice::Truncate)); bad.write("not a png at all, just bytes"); bad.close(); }
        Harness h; QVERIFY(h.window); h.window->resize(1180, 780);
        QVERIFY(openFolder(h, media)); QVERIFY(activate(h)); focusCanvas(h);
        QTRY_COMPARE(h.viewer.currentIndex(), 1);
        h.viewer.setBoundaryMode("loop");
        QVERIFY(QMetaObject::invokeMethod(h.window, "setSlideshowSeconds", Q_ARG(QVariant, 1)));
        h.viewer.trigger("slideshow");
        QTRY_VERIFY(flag(h, "slideshowOn"));
        QStringList trace;
        for (int s = 0; s < 8; ++s) {
            QTest::qWait(1000);
            trace << QString("t=%1s index=%2 ready=%3 loading=%4 on=%5 status=\"%6\"").arg(s + 1).arg(h.viewer.currentIndex())
                     .arg(h.viewer.imageReady()).arg(h.viewer.loading()).arg(flag(h, "slideshowOn")).arg(h.viewer.status());
        }
        for (const auto& line : trace) qInfo("%s", qPrintable(line));
        QVERIFY2(h.viewer.currentIndex() != 2 || !flag(h, "slideshowOn"), "slideshow stays on, stuck at the undecodable photo");
    }

    void probeDetailsToggleKeepsVisibleAnchor() {
        Media media(120, QSize(32, 24)); Harness h; QVERIFY(h.window); h.window->resize(1180, 780);
        QVERIFY(openFolder(h, media)); QVERIFY(activate(h));
        h.viewer.trigger("grid"); QTRY_VERIFY(flag(h, "gridMode")); QTRY_COMPARE(h.fileGrid.count(), 120);
        auto* view = named(h.window, "photoGridView"); QVERIFY(view);
        auto firstVisible = [&] {
            const qreal y = view->property("contentY").toReal() + 4;
            int index = -1;
            QMetaObject::invokeMethod(view, "indexAt", Q_RETURN_ARG(int, index), Q_ARG(qreal, 8), Q_ARG(qreal, y));
            return index;
        };
        auto inView = [&](int row) {
            for (auto* c : namedAll(h.window, "photoGridCell")) if (c->property("index").toInt() == row) {
                const QPointF top = c->mapToItem(view, QPointF(0, 0));
                return top.y() + c->height() > 0 && top.y() < view->height();
            }
            return false;
        };
        QTest::qWait(400);
        // Case A: wheel-style scroll to the middle, cursor left at the top (off screen).
        const qreal mid = (view->property("contentHeight").toReal() - view->height()) / 2;
        view->setProperty("contentY", view->property("originY").toReal() + mid); QTest::qWait(300);
        const int beforeA = firstVisible();
        h.viewer.trigger("view-details"); QTRY_VERIFY(h.object("photoGrid")->property("details").toBool()); QTest::qWait(400);
        const int afterA = firstVisible();
        qInfo("case A (scrolled, cursor off screen): first visible before %d, after details %d", beforeA, afterA);
        h.viewer.trigger("view-details"); QTRY_VERIFY(!h.object("photoGrid")->property("details").toBool()); QTest::qWait(400);
        const int backA = firstVisible();
        qInfo("case A: first visible after returning to thumbnails %d", backA);
        // Case B: keyboard cursor on a visible item in the middle.
        view->setProperty("contentY", view->property("originY").toReal() + mid); QTest::qWait(300);
        const int cursor = firstVisible() + 2; h.fileGrid.select(cursor, DirectoryEntryModel::Replace); QTest::qWait(300);
        const bool cursorBefore = inView(cursor);
        h.viewer.trigger("view-details"); QTRY_VERIFY(h.object("photoGrid")->property("details").toBool()); QTest::qWait(400);
        qInfo("case B (cursor %d visible=%d): after details first visible %d, cursor visible=%d", cursor, cursorBefore, firstVisible(), inView(cursor));
        h.viewer.trigger("view-details"); QTest::qWait(400);
        qInfo("case B: back to thumbnails, first visible %d, cursor visible=%d", firstVisible(), inView(cursor));
        for (const int width : {1180, 1800, 760}) {
            h.window->resize(width, 780); QTest::qWait(500);
            const qreal m = (view->property("contentHeight").toReal() - view->height()) / 5;
            view->setProperty("contentY", view->property("originY").toReal() + m); QTest::qWait(300);
            const int b = firstVisible();
            const int columns = h.object("photoGrid")->property("columns").toInt();
            h.viewer.trigger("view-details"); QTRY_VERIFY(h.object("photoGrid")->property("details").toBool()); QTest::qWait(400);
            const int a = firstVisible();
            qInfo("width %d (%d columns): first visible before %d, after details %d", width, columns, b, a);
            h.viewer.trigger("view-details"); QTRY_VERIFY(!h.object("photoGrid")->property("details").toBool()); QTest::qWait(400);
        }
        QVERIFY2(qAbs(afterA - beforeA) <= 1, "details toggle moved the visible anchor");
    }
