class PreviewProvider final : public QQuickImageProvider {
public:
    PreviewProvider() : QQuickImageProvider(Image) {}
    std::atomic_int requests{0};
    std::atomic_bool succeed{false};
    QMutex mutex;
    QStringList paths; // every decoded request path, to prove non-images never reach the thumbnail pipeline
    QStringList requested() { QMutexLocker lock(&mutex); return paths; }
    QImage requestImage(const QString& id, QSize* size, const QSize& requested) override {
        ++requests;
        const auto parts = thumbnail_id::decode(id);
        { QMutexLocker lock(&mutex); paths << (parts ? parts->path : QStringLiteral("<malformed>")); }
        if (size) *size = QSize(32, 24);
        Q_UNUSED(requested);
        // By default previews fail deliberately; each cell must remain named and activatable. Review captures use
        // generated previews instead, except for names containing "broken".
        if (!succeed || (parts && parts->path.contains("broken"))) return {};
        QImage image(160, 120, QImage::Format_RGB32);
        const uint seed = qHash(parts ? parts->path : id);
        image.fill(QColor::fromHsv(int(seed % 360), 90, 170));
        for (int y = 0; y < 120; ++y) for (int x = 0; x < 160; x += 1) if (y > 70 + int(20 * std::sin(x / 18.0 + seed))) image.setPixelColor(x, y, QColor::fromHsv(int((seed / 7) % 360), 120, 90));
        if (size) *size = image.size();
        return image;
    }
};

// Disposable folders of empty "Photo N.png" files for tests that only need a listing; shared across cases so the
// 10,000-entry fixture is generated once.
QTemporaryDir* listingFixture(int count) {
    static std::map<int, std::unique_ptr<QTemporaryDir>> fixtures;
    if (count < 0) { fixtures.clear(); return nullptr; }
    auto& slot = fixtures[count];
    if (!slot) {
        slot = std::make_unique<QTemporaryDir>();
        for (int i = 0; i < count; ++i) { QFile file(slot->filePath(QString("Photo %1.png").arg(i))); if (file.open(QIODevice::WriteOnly)) file.close(); }
    }
    return slot.get();
}

// DELETE-01: the UI harness never touches the real Trash/Recycle Bin. This backend renames (never copies or deletes)
// into a disposable folder, refuses anything outside the temp directory, can fail chosen file names and can be held.
struct TestTrash final : glancelot::TrashBackend {
    std::filesystem::path bin;
    std::mutex mutex; std::condition_variable wake; bool held{}; int waiting{};
    std::map<QString, glancelot::DeleteOutcome> fail; // by file name
    std::atomic_int calls{0};
    explicit TestTrash(const QString& folder) : bin(folder.toStdWString()) {}
    void hold() { std::lock_guard lock(mutex); held = true; }
    void release() { { std::lock_guard lock(mutex); held = false; } wake.notify_all(); }
    int blocked() { std::lock_guard lock(mutex); return waiting; }
    glancelot::DeleteResult moveToTrash(const glancelot::MutationTarget& target) override {
        { std::unique_lock lock(mutex); ++waiting; wake.wait(lock, [this] { return !held; }); --waiting; }
        const int n = ++calls;
        const QString name = QString::fromStdWString(target.path.filename().wstring());
        const QString temp = QDir::cleanPath(QDir::tempPath()).toCaseFolded();
        if (!QDir::cleanPath(QString::fromStdWString(target.path.wstring()).replace('\\', '/')).toCaseFolded().startsWith(temp))
            return {target.path, glancelot::DeleteOutcome::Denied, "test trash only accepts generated files"};
        { std::lock_guard lock(mutex); if (auto it = fail.find(name); it != fail.end()) return {target.path, it->second, "simulated"}; }
        std::error_code ec;
        std::filesystem::rename(target.path, bin / (std::to_wstring(n) + L"-" + target.path.filename().wstring()), ec);
        return {target.path, ec ? glancelot::DeleteOutcome::Failed : glancelot::DeleteOutcome::Trashed, ec ? ec.message() : std::string{}};
    }
};

struct Harness {
    QTemporaryDir config;
    ViewerController viewer;
    ThemeManager theme;
    FilmstripModel model;
    FolderTreeModel folders;
    PaneLayout layout;
    DirectoryEntryModel fileGrid;
    QTemporaryDir trashBin;
    TestTrash* trash = new TestTrash(trashBin.path());
    FileOperationService fileOperations{std::unique_ptr<glancelot::TrashBackend>(trash)};
    QQmlApplicationEngine engine;
    PreviewProvider* provider = new PreviewProvider;
    QQuickWindow* window{};
    explicit Harness(bool live = false, int slot = 0) {
        if (slot > 0) layout.setSlot(slot); // as app/main.cpp: the slot's layout is loaded before the UI
        // Load the UI from source: drop the executable's directory so a build-tree "Glancelot" module cannot shadow it.
        QStringList imports = engine.importPathList();
        imports.removeAll(QCoreApplication::applicationDirPath());
        engine.setImportPathList(imports);
        engine.addImageProvider("thumb", provider);
        engine.rootContext()->setContextProperty("viewer", &viewer);
        engine.rootContext()->setContextProperty("paneLayout", &layout);
        engine.rootContext()->setContextProperty("theme", &theme);
        engine.rootContext()->setContextProperty("filmstrip", &model);
        engine.rootContext()->setContextProperty("folderTree", &folders);
        engine.rootContext()->setContextProperty("fileGrid", &fileGrid);
        engine.rootContext()->setContextProperty("fileOperations", &fileOperations);
        file_operation_wiring::follow(fileOperations, viewer, folders); // the shipped reconciliation wiring
        // Tests that inject a listing with fill() must not be overwritten by the (empty) controller snapshot.
        // Live wiring is app/main.cpp's: the same file_grid_wiring helpers, plus the filmstrip connections.
        if (live) {
            QObject::connect(&viewer, &ViewerController::stateChanged, &model, [&] { model.sync(viewer.snapshot()); });
            QObject::connect(&viewer, &ViewerController::filesInvalidated, &model, [&] { model.invalidate(); });
            file_grid_wiring::followViewer(viewer, fileGrid);
        }
        file_grid_wiring::lockThumbnailEpoch(model, fileGrid);
        engine.load(QUrl::fromLocalFile(QStringLiteral(GLANCELOT_TEST_SOURCE_DIR "/ui/Main.qml")));
        if (!engine.rootObjects().empty()) window = qobject_cast<QQuickWindow*>(engine.rootObjects().first());
    }
    QQuickItem* grid() const { return window->findChild<QQuickItem*>("photoGrid"); }
    QQuickItem* view() const { return window->findChild<QQuickItem*>("photoGridView"); }
    void showGrid() { viewer.trigger("grid"); window->requestActivate(); }
    // Injects a real generated listing into the grid (non-live harness) with `current` as the viewed photo.
    void fill(int count, int current = 0) {
        QTemporaryDir* dir = listingFixture(count);
        fileGrid.setCurrentImage(count ? dir->filePath(QString("Photo %1.png").arg(current)) : QString{});
        fileGrid.setDirectory(dir->path());
    }
    int selected() const { return grid()->property("selectedIndex").toInt(); }
    // Delegates are visual (not QObject) children, so walk the item tree.
    QList<QQuickItem*> cells() const {
        QList<QQuickItem*> found; QList<QQuickItem*> pending{view()};
        while (!pending.isEmpty()) {
            QQuickItem* item = pending.takeFirst();
            if (item->objectName() == "photoGridCell") found << item;
            pending << item->childItems();
        }
        return found;
    }
    QQuickItem* cell(int index) const { for (auto* c : cells()) if (c->property("index").toInt() == index) return c; return nullptr; }
};

// TOPBAR-THEME-01 helpers: find a named item anywhere in the visual tree, and WCAG contrast.
static QQuickItem* namedItem(QQuickWindow* window, const QString& name) {
    QList<QQuickItem*> pending{window->contentItem()};
    while (!pending.isEmpty()) {
        QQuickItem* item = pending.takeFirst();
        if (item->objectName() == name) return item;
        pending << item->childItems();
    }
    return nullptr;
}
static double relativeLuminance(const QColor& color) {
    const auto channel = [](double v) { return v <= 0.04045 ? v / 12.92 : std::pow((v + 0.055) / 1.055, 2.4); };
    return 0.2126 * channel(color.redF()) + 0.7152 * channel(color.greenF()) + 0.0722 * channel(color.blueF());
}
static double contrastRatio(const QColor& a, const QColor& b) {
    const double x = relativeLuminance(a), y = relativeLuminance(b);
    return (std::max(x, y) + 0.05) / (std::min(x, y) + 0.05);
}
static QColor over(const QColor& top, const QColor& below) { // source-over compositing onto an opaque color
    const double a = top.alphaF();
    return QColor::fromRgbF(top.redF() * a + below.redF() * (1 - a), top.greenF() * a + below.greenF() * (1 - a), top.blueF() * a + below.blueF() * (1 - a));
}
static QColor itemColor(QQuickItem* item, const char* property = "color") { return item ? item->property(property).value<QColor>() : QColor(); }
static QQuickItem* buttonBackground(QQuickItem* button) { return button ? button->property("background").value<QQuickItem*>() : nullptr; }
static QQuickItem* buttonIcon(QQuickItem* button) {
    QList<QQuickItem*> pending{button};
    while (!pending.isEmpty()) {
        QQuickItem* item = pending.takeFirst();
        if (item != button && item->property("paths").isValid() && item->property("name").toString().size()) return item;
        pending << item->childItems();
    }
    return nullptr;
}

    // Visible items that leave the window, and visible Text that is clipped by its own width.
    static QStringList probe(QQuickWindow* w) {
        QStringList issues; QList<QQuickItem*> pending{w->contentItem()};
        const QRectF win(0, 0, w->width(), w->height());
        while (!pending.isEmpty()) {
            QQuickItem* item = pending.takeFirst();
            pending << item->childItems();
            if (!item->isVisible() || item->opacity() <= 0.01 || item->width() <= 0 || item->height() <= 0) continue;
            const QString cls = item->metaObject()->className();
            if (cls.startsWith("QQuickText")) {
                const qreal implicit = item->property("implicitWidth").toReal();
                const int elide = item->property("elide").toInt();
                const QString text = item->property("text").toString();
                if (!text.isEmpty() && elide == 0 && item->property("wrapMode").toInt() == 0 && implicit > item->width() + 1) {
                    const QRectF r(item->mapToScene(QPointF(0, 0)), QSizeF(item->width(), item->height()));
                    if (win.intersects(r)) issues << QString("clipped-text '%1' implicit=%2 width=%3").arg(text.left(40)).arg(implicit).arg(item->width());
                }
            }
            const QRectF r(item->mapToScene(QPointF(0, 0)), QSizeF(item->width(), item->height()));
            if (!item->objectName().isEmpty() && (r.right() > win.right() + 1 || r.bottom() > win.bottom() + 1 || r.left() < -1 || r.top() < -1) && win.intersects(r))
                issues << QString("outside-window %1 (%2) rect=%3,%4 %5x%6").arg(item->objectName(), cls).arg(r.x()).arg(r.y()).arg(r.width()).arg(r.height());
        }
        return issues;
    }

    // Full worker protocol: probe, shared-memory handshake, decoded, ok. Crash or hang fails.
    void decoderProtocolFuzz() {
        QString program = QCoreApplication::applicationDirPath() + "/glancelot-decoder.exe";
        QVERIFY2(QFileInfo::exists(program), qPrintable(program));
        QTemporaryDir dir; QVERIFY(dir.isValid());
        QList<QByteArray> seeds; QStringList exts{"png", "jpg", "bmp", "tif"};
        for (const QString& ext : exts) { QImage img(160, 120, QImage::Format_RGB32); img.fill(Qt::darkCyan); QBuffer buf; buf.open(QIODevice::WriteOnly); QVERIFY(img.save(&buf, ext.toLatin1().constData())); seeds << buf.data(); }
        QRandomGenerator rng(4242); int decoded = 0, rejected = 0, bad = 0; const int total = 400;
        for (int n = 0; n < total; ++n) {
            QByteArray data = seeds[n % seeds.size()];
            const int mode = n % 4;
            if (mode == 0) { for (int k = 0, c = 1 + rng.bounded(10); k < c; ++k) data[rng.bounded(data.size())] = char(rng.bounded(256)); }
            else if (mode == 1) data.truncate(rng.bounded(1, data.size()));
            else if (mode == 2) { for (int k = 0; k < 30; ++k) data[rng.bounded(qMin(data.size(), 64))] = char(rng.bounded(256)); }
            else { const int at = rng.bounded(data.size()); data.insert(at, QByteArray(rng.bounded(1, 64), char(rng.bounded(256)))); }
            const QString path = dir.filePath(QString("f%1.%2").arg(n).arg(exts[n % exts.size()]));
            QFile f(path); QVERIFY(f.open(QIODevice::WriteOnly)); f.write(data); f.close();
            QProcess p; p.setProgram(program); p.setArguments({path}); p.setStandardErrorFile(QProcess::nullDevice()); p.start();
            if (!p.waitForStarted(5000)) { ++bad; qWarning() << "AUDIT-FUZZ start failed" << path; continue; }
            std::unique_ptr<QSharedMemory> shm; QByteArray out; bool done = false, hang = false;
            QElapsedTimer t; t.start();
            while (!done) {
                if (!p.waitForReadyRead(200) && p.state() == QProcess::NotRunning) { out += p.readAllStandardOutput(); break; }
                out += p.readAllStandardOutput();
                while (out.contains('\n')) {
                    const int nl = out.indexOf('\n'); const auto line = out.left(nl); out.remove(0, nl + 1);
                    const auto obj = QJsonDocument::fromJson(line).object(); const auto phase = obj.value("phase").toString();
                    if (phase == "probe") {
                        const qint64 bytes = obj.value("bytes").toInteger();
                        if (bytes <= 0 || bytes > 64 * 1024 * 1024) { done = true; ++rejected; break; }
                        shm = std::make_unique<QSharedMemory>("Glancelot-fuzz-" + QUuid::createUuid().toString(QUuid::WithoutBraces));
                        if (!shm->create(bytes)) { done = true; break; }
                        p.write(QJsonDocument(QJsonObject{{"key", shm->key()}}).toJson(QJsonDocument::Compact) + '\n');
                    } else if (phase == "decoded") { ++decoded; p.write("ok\n"); done = true; break; }
                    else if (phase == "error") { ++rejected; done = true; break; }
                }
                if (t.elapsed() > 8000) { hang = true; break; }
            }
            if (hang) { ++bad; qWarning() << "AUDIT-FUZZ HANG" << path; p.kill(); }
            p.waitForFinished(3000);
            if (!hang && p.state() == QProcess::NotRunning && p.exitStatus() == QProcess::CrashExit) { ++bad; qWarning() << "AUDIT-FUZZ CRASH" << path << p.exitCode(); }
            if (p.state() != QProcess::NotRunning) p.kill();
        }
        qInfo() << "AUDIT-FUZZ total" << total << "decoded" << decoded << "rejected" << rejected << "bad" << bad;
        QCOMPARE(bad, 0);
    }

    void animationFuzz() {
        const QString dir = qEnvironmentVariable("AUDIT_GIF_DIR"); QVERIFY(!dir.isEmpty());
        QList<QByteArray> seeds; for (const QString n : {"static.gif", "anim.gif"}) { QFile f(dir + "/" + n); QVERIFY(f.open(QIODevice::ReadOnly)); seeds << f.readAll(); }
        QTemporaryDir out; QRandomGenerator rng(99); int loadedOk = 0, total = 150;
        glancelot::AnimationController anim;
        for (int i = 0; i < total; ++i) {
            QByteArray d = seeds[i % 2];
            switch (i % 4) { case 0: for (int k = 0, c = 1 + rng.bounded(8); k < c; ++k) d[rng.bounded(d.size())] = char(rng.bounded(256)); break;
                case 1: d.truncate(rng.bounded(1, d.size())); break;
                case 2: for (int k = 0; k < 30; ++k) d[rng.bounded(qMin(d.size(), 40))] = char(rng.bounded(256)); break;
                default: d.insert(rng.bounded(d.size()), QByteArray(rng.bounded(1, 50), char(rng.bounded(256)))); }
            const QString path = out.filePath(QString("m%1.gif").arg(i)); QFile f(path); QVERIFY(f.open(QIODevice::WriteOnly)); f.write(d); f.close();
            QSignalSpy spy(&anim, &glancelot::AnimationController::loaded);
            anim.load(path, quint64(i + 1));
            QVERIFY2(spy.wait(8000) || spy.count() > 0, qPrintable(QString("no loaded signal for %1").arg(path)));
            if (anim.frameCount() > 0) { ++loadedOk; anim.jumpToFrame(anim.frameCount() / 2); anim.nextFrame(); anim.previousFrame(); }
            anim.clear();
        }
        qInfo() << "AUDIT-ANIM-FUZZ total" << total << "with-frames" << loadedOk;
    }
    void animationMemory() {
        const QString path = qEnvironmentVariable("AUDIT_BIG_GIF"); QVERIFY(QFileInfo::exists(path));
        glancelot::AnimationController anim; QSignalSpy spy(&anim, &glancelot::AnimationController::loaded);
        PROCESS_MEMORY_COUNTERS pmc{}; GetProcessMemoryInfo(GetCurrentProcess(), &pmc, sizeof(pmc)); const SIZE_T before = pmc.PeakWorkingSetSize;
        QElapsedTimer t; t.start(); anim.load(path, 1); const bool ok = spy.wait(120000);
        GetProcessMemoryInfo(GetCurrentProcess(), &pmc, sizeof(pmc));
        qInfo() << "AUDIT-ANIM-MEM loaded" << ok << "frames" << anim.frameCount() << "ms" << t.elapsed() << "peakMB" << (pmc.PeakWorkingSetSize >> 20) << "beforeMB" << (before >> 20);
    }
